# Ghép Câu MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the web MVP of "Ghép Câu" — a mobile-first English-communication learning game for Vietnamese office workers, one scene ("Sếp hỏi task"), matching English chunks ↔ Vietnamese meanings that assemble into a full message the app speaks aloud.

**Architecture:** Single Next.js App-Router client app. Content lives in a verbatim JSON file typed by `content/types.ts`. A framework-agnostic state machine (`lib/useGame.ts`) drives five screens. Pure logic (state machine, tile matching, scoring, name handling, storage) is unit-tested with Vitest; visual/UI is verified via the running app + screenshots. No backend, DB, or API routes.

**Tech Stack:** Next.js (App Router), TypeScript, Tailwind CSS, Vitest, `next/font/google` (Baloo 2 + Nunito), Web Speech API (TTS), Web Audio API (SFX), `localStorage`.

## Global Constraints

Every task's requirements implicitly include this section. Source of truth = the user's spec (mục 0–13) + design doc `docs/superpowers/specs/2026-07-21-ghep-cau-design.md`.

- **Verbatim Vietnamese copy** — never rewrite, translate, "fix", or normalize any Vietnamese string. Văn nói (`ời`, `nhe`, `hen`, `haha`, `=))))`, emoji) is intentional. Copy every character exactly.
- **No new learning content** — no extra sentences, scenes, variants, or grammar notes. Only the data in the JSON.
- **Palette is frozen** (role-named): `bg-page #FFFCF2`, `bg-surface #FFFFFF`, `bg-sunken #F2ECE0`, `ink #2B2118`, `ink-soft #8A7F6E`, `en #FF7A00`, `vi #00B2A9`, `mastered #FFD029`, `ok #7ED957`, `no #FF4D3D`.
- **`mastered` (yellow) is used ONLY for combo** (Decision D1/D3). Not avatars, not buttons, not matched cells, not boss options.
- **No gradient, no glow/neon, no soft box-shadow, no blur** — anywhere. Sticker style: cream background, solid color blocks, thick black borders.
- **Shape constants:** border `2px solid ink` (primary buttons `3px`); radius tile/card/bubble `14px`, button/chip `20px`; button sunk-lip `4px` (20% darker than button fill), `:active` drops `4px` / lip `0`.
- **Type scale (px):** 11 / 12.5 / 14 / 15.5 / 18 / 22 / 32. Display = Baloo 2 (700/800); Body = Nunito (400/600). `subsets: ['latin','vietnamese']`.
- **Mobile-first:** design for 390px width; desktop centers a `max-width: 520px` column.
- **Content in JSON only** — components import strings from the JSON, never inline them.
- **TTS always strips `{name}`** before speaking. Never crash when Web Speech / Web Audio / localStorage is unavailable.
- **Respect `prefers-reduced-motion`** (disable particles, keep color transitions).
- No backend, no DB, no API routes.

**Decisions (from brainstorming, see design doc):**
- **D1:** Boss option the player wrongly picks (`ok:false`) → `ink-soft` **dashed** border (neutral), NOT yellow. `ok:true` → `ok` (green) border. Yellow reserved for combo only.
- **D2:** SummaryScreen `n/3 đúng tông` = count of BossScreen questions answered correctly (0–3).
- **D4:** Score = one running total: Σ(`10×combo` per correct match) + Σ(`+50` per correct boss). `gc:bestScore` = max of this total.

```ts
const K = { name: "gc:name", best: "gc:bestScore", done: "gc:completed" } as const;

function read(key: string): string | null {
  try { return localStorage.getItem(key); } catch { return null; }
}
function write(key: string, val: string): void {
  try { localStorage.setItem(key, val); } catch { /* ignore */ }
}

export function getName(): string | null { return read(K.name); }
export function setName(name: string): void { write(K.name, name); }

export function getBestScore(): number { return Number(read(K.best) ?? 0) || 0; }
export function setBestScore(n: number): void { if (n > getBestScore()) write(K.best, String(n)); }

export function getCompleted(): string[] {
  try { return JSON.parse(read(K.done) ?? "[]"); } catch { return []; }
}
export function addCompleted(id: string): void {
  const set = new Set(getCompleted()); set.add(id);
  write(K.done, JSON.stringify([...set]));
}
```

export function stripName(text: string): string {
  return text
    .replace(/\{name\}/g, "")
    .replace(/^\s*[,—-]\s*/, "")   // drop leading comma/dash left by removed name
    .replace(/\s{2,}/g, " ")
    .trim();
}

function pickVoice(): SpeechSynthesisVoice | null {
  const voices = window.speechSynthesis.getVoices();
  return voices.find(v => v.lang === "en-US")
      ?? voices.find(v => v.lang?.startsWith("en"))
      ?? null;
}

export function initVoices(): void {
  if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
  window.speechSynthesis.onvoiceschanged = () => { /* voices now cached */ };
  window.speechSynthesis.getVoices();
}

export function speak(text: string, opts: { rate?: number } = {}): void {
  if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
  try {
    const synth = window.speechSynthesis;
    synth.cancel();
    const u = new SpeechSynthesisUtterance(stripName(text));
    u.lang = "en-US";
    u.rate = opts.rate ?? 1.0;
    const v = pickVoice();
    if (v) u.voice = v;
    synth.speak(u);
  } catch { /* ignore */ }
}

let ctx: AudioContext | null = null;

function AC(): typeof AudioContext | null {
  if (typeof window === "undefined") return null;
  return window.AudioContext ?? (window as any).webkitAudioContext ?? null;
}

export function initAudio(): void {
  try {
    const Ctor = AC();
    if (!Ctor) return;
    if (!ctx) ctx = new Ctor();
    if (ctx.state === "suspended") ctx.resume();
  } catch { /* ignore */ }
}

function tone(freq: number, dur: number, type: OscillatorType = "square", gain = 0.06, delay = 0): void {
  try {
    if (!ctx) return;
    const t0 = ctx.currentTime + delay;
    const osc = ctx.createOscillator();
    const g = ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, t0);
    g.gain.setValueAtTime(gain, t0);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
    osc.connect(g).connect(ctx.destination);
    osc.start(t0);
    osc.stop(t0 + dur);
  } catch { /* ignore */ }
}

```ts
import type { Chunk } from "@/content/types";

export type TileModel = { key: string; chunkId: number; side: "en" | "vi"; text: string };
export type TapResult = { kind: "select" | "deselect" | "reselect" | "match" | "miss" };

export function buildTiles(chunks: Chunk[], rand: () => number = Math.random): TileModel[] {
  const tiles: TileModel[] = [];
  chunks.forEach((c, i) => {
    tiles.push({ key: `en-${i}`, chunkId: i, side: "en", text: c.en });
    tiles.push({ key: `vi-${i}`, chunkId: i, side: "vi", text: c.vi });
  });
  for (let i = tiles.length - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1));
    [tiles[i], tiles[j]] = [tiles[j], tiles[i]];
  }
  return tiles;
}

export function resolveTap(selected: TileModel | null, tapped: TileModel): TapResult {
  if (!selected) return { kind: "select" };
  if (selected.key === tapped.key) return { kind: "deselect" };
  if (selected.side === tapped.side) return { kind: "reselect" };
  return selected.chunkId === tapped.chunkId ? { kind: "match" } : { kind: "miss" };
}

"use client";
import { useCallback, useEffect, useState } from "react";
import type { Scene } from "@/content/types";
import * as storage from "./storage";

export type Screen = "name" | "intro" | "play" | "boss" | "summary";
export type GameState = {
  screen: Screen; name: string;
  lineIndex: number; bossIndex: number;
  score: number; bestScore: number;
  bossCorrect: number; maxCombo: number; combo: number;
};

export function initState(scene: Scene, storedName: string | null, storedBest: number): GameState {
  return {
    screen: storedName ? "intro" : "name",
    name: storedName ?? "bạn",
    lineIndex: 0, bossIndex: 0,
    score: 0, bestScore: storedBest,
    bossCorrect: 0, maxCombo: 0, combo: 0,
  };
}

export function applyName(s: GameState, raw: string): GameState {
  const name = raw.trim().slice(0, 14) || "bạn";
  storage.setName(name);
  return { ...s, name, screen: "intro" };
}

export function applyMatch(s: GameState): GameState {
  const combo = s.combo + 1;
  return { ...s, combo, score: s.score + 10 * combo, maxCombo: Math.max(s.maxCombo, combo) };
}
export function applyMiss(s: GameState): GameState { return { ...s, combo: 0 }; }

export function nextLine(s: GameState, scene: Scene): GameState {
  const next = s.lineIndex + 1;
  if (next >= scene.lines.length) return { ...s, screen: "boss", combo: 0 };
  return { ...s, lineIndex: next, combo: 0 };
}

export function answerBoss(s: GameState, correct: boolean): GameState {
  return correct ? { ...s, score: s.score + 50, bossCorrect: s.bossCorrect + 1 } : s;
}

export function nextBoss(s: GameState, scene: Scene): GameState {
  const next = s.bossIndex + 1;
  if (next >= scene.boss.length) {
    storage.setBestScore(s.score);
    storage.addCompleted(scene.id);
    return { ...s, screen: "summary", bestScore: Math.max(s.bestScore, s.score) };
  }
  return { ...s, bossIndex: next };
}

export function withName(text: string, name: string): string {
  return text.replace(/\{name\}/g, name);
}

export function useGame(scene: Scene) {
  const [state, setState] = useState<GameState>(() => initState(scene, null, 0));
  useEffect(() => { setState(initState(scene, storage.getName(), storage.getBestScore())); }, [scene]);

  return {
    state,
    setName: useCallback((raw: string) => setState(s => applyName(s, raw)), []),
    startScene: useCallback(() => setState(s => ({ ...s, screen: "play" as Screen })), []),
    match: useCallback(() => setState(applyMatch), []),
    miss: useCallback(() => setState(applyMiss), []),
    advanceLine: useCallback(() => setState(s => nextLine(s, scene)), [scene]),
    answerBoss: useCallback((ok: boolean) => setState(s => answerBoss(s, ok)), []),
    advanceBoss: useCallback(() => setState(s => nextBoss(s, scene)), [scene]),
    replay: useCallback(() => setState(s => ({ ...initState(scene, s.name, Math.max(s.bestScore, s.score)), screen: "intro" as Screen })), [scene]),
  };
}
```

Behaviour (spec 7.3–7.4, mục 8/10), timings exact:
1. On new line: if `them != null` → set status `đang soạn tin trên máy tính…`, show `TypingIndicator` **1500ms**; then remove, status `đang hoạt động`, push a `them` bubble (`withName(en)` + `withName(vi)`), `sfx.ping()`, `speak(withName-stripped en)`. Wait **900ms**. If `them == null`, skip these.

3. 4. When all chunks filled → send: Composer label `Đang gửi…`, `speak(line.en, {rate:0.9})`, sequentially set `glowIndex` per cell with duration ∝ chunk char length; on done hide Board+Composer, push `me` bubble (`line.en`, `seen:true`, `onSpeak: ()=>speak(line.en,{rate:0.9})`), `sfx.send()`.

- [ ] **Step 1: Implement PlayScreen.tsx** (full component)

```tsx
"use client";
import { useEffect, useMemo, useRef, useState } from "react";
import Board from "../game/Board";
import Composer from "../game/Composer";
import Chip from "../ui/Chip";
import Button from "../ui/Button";
import ComboBadge from "../effects/ComboBadge";
import Particles from "../effects/Particles";
import TypingIndicator from "../game/TypingIndicator";
import { buildTiles, resolveTap, type TileModel } from "@/lib/matching";
import { speak } from "@/lib/speech";
import { sfx } from "@/lib/sfx";
import { withName, type GameState } from "@/lib/useGame";
import type { Scene } from "@/content/types";
import type { ChatEntry } from "../game/ChatThread";

type Phase = "intro" | "think" | "play" | "sending" | "done";

export default function PlayScreen({
  scene, state, onMatch, onMiss, onAdvance, onExit, chat, pushChat, ChatSlot,
}: {
  scene: Scene; state: GameState;
  onMatch: () => void; onMiss: () => void; onAdvance: () => void; onExit: () => void;
  chat: ChatEntry[]; pushChat: (e: ChatEntry) => void; ChatSlot: React.ReactNode;
}) {
  const line = scene.lines[state.lineIndex];
  const [phase, setPhase] = useState<Phase>("intro");
  const [status, setStatus] = useState("đang hoạt động");
  const [tiles, setTiles] = useState<TileModel[]>([]);
  const [filled, setFilled] = useState<boolean[]>(() => line.chunks.map(() => false));
  const [selected, setSelected] = useState<TileModel | null>(null);
  const [shaking, setShaking] = useState<string[]>([]);
  const [locked, setLocked] = useState(true);
  const [glowIndex, setGlowIndex] = useState<number | undefined>(undefined);
  const [burst, setBurst] = useState<{ x: number; y: number; color: string } | null>(null);
  const [variantOpen, setVariantOpen] = useState(false);
  const comboRef = useRef(state.combo);
  comboRef.current = state.combo;

  // reset when line changes
  useEffect(() => {
    setPhase("intro"); setFilled(line.chunks.map(() => false));
    setSelected(null); setLocked(true); setVariantOpen(false);
    setTiles(buildTiles(line.chunks));
    const timers: number[] = [];
    const startPlay = () => {
      pushChatThink();
      timers.push(window.setTimeout(() => { setPhase("play"); setLocked(false); }, 600));
    };
    const pushChatThink = () => setPhase("think");
    if (line.them) {
      setStatus("đang soạn tin trên máy tính…");
      setPhase("intro");
      timers.push(window.setTimeout(() => {
        setStatus("đang hoạt động");
        pushChat({ id: `them-${state.lineIndex}`, kind: "them",
          en: withName(line.them!.en, state.name), vi: withName(line.them!.vi, state.name) });
        sfx.ping(); speak(line.them!.en, { rate: 1 });
        timers.push(window.setTimeout(startPlay, 900));
      }, 1500));
    } else {
      startPlay();
    }
    return () => timers.forEach(clearTimeout);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [state.lineIndex]);

  const remaining = useMemo(() => tiles, [tiles]);

  function onTap(t: TileModel) {
    if (locked) return;
    const r = resolveTap(selected, t);
    if (r.kind === "select") { setSelected(t); sfx.click(); return; }
    if (r.kind === "deselect") { setSelected(null); return; }
    if (r.kind === "reselect") { setSelected(t); sfx.click(); return; }
    if (r.kind === "match") {
      setLocked(true);
      onMatch();
      sfx.matchOk(comboRef.current + 1);
      speak(t.text.length ? (t.side === "en" ? t.text : selected!.text) : t.text, { rate: 1 });
      const el = document.querySelector(`[data-key="${t.key}"]`) as HTMLElement | null;
      if (el) { const r2 = el.getBoundingClientRect(); setBurst({ x: r2.left + r2.width / 2, y: r2.top + r2.height / 2, color: "#FFD029" }); }
      const id = t.chunkId;
      setTiles(prev => prev.filter(x => x.chunkId !== id));
      setFilled(prev => prev.map((f, i) => (i === id ? true : f)));
      setSelected(null);
      window.setTimeout(() => { setBurst(null); setLocked(false); }, 400);
    } else {
      setLocked(true);
      onMiss();
      sfx.matchNo();
      navigator.vibrate?.(60);
      setShaking([t.key, selected!.key]);
      window.setTimeout(() => { setShaking([]); setSelected(null); setLocked(false); }, 420);
    }
  }

  // trigger send when all filled
  useEffect(() => {
    if (phase === "play" && filled.length && filled.every(Boolean)) runSend();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [filled, phase]);

  function runSend() {
    setPhase("sending");
    speak(line.en, { rate: 0.9 });
    let acc = 0;
    line.chunks.forEach((c, i) => {
      const dur = Math.max(280, c.en.length * 28);
      window.setTimeout(() => setGlowIndex(i), acc);
      acc += dur;
    });
    window.setTimeout(() => {
      setGlowIndex(undefined);
      pushChat({ id: `me-${state.lineIndex}`, kind: "me", en: line.en, seen: true,
        onSpeak: () => speak(line.en, { rate: 0.9 }) });
      sfx.send();
      setPhase("done");
    }, acc + 200);
  }

  const showBoard = phase === "play";
  const showComposer = phase === "play" || phase === "sending";

  return (
    <div className="mx-auto max-w-[520px] px-4 pb-8 flex flex-col gap-3">
      {/* top bar */}
      <div className="flex items-center gap-2 pt-3">
        <button onClick={onExit} aria-label="Thoát" className="text-lg font-display px-2">✕</button>
        <div className="w-9 h-9 rounded-full bg-sunken border-2 border-ink flex items-center justify-center">🧑‍💼</div>
        <div className="flex-1 leading-tight">
          <p className="font-display font-bold text-sm">Sếp</p>
          <p className="text-2xs text-ink-soft">{status}</p>
        </div>
        <Chip>Lượt {state.lineIndex + 1}/3</Chip>
        <Chip>Điểm {state.score}</Chip>
      </div>

      {line.sit !== "" && (
        <p className="italic text-xs text-ink-soft bg-sunken border-2 border-dashed border-ink rounded-card px-3 py-2">{line.sit}</p>
      )}

      {ChatSlot}
      {phase === "intro" && line.them && <TypingIndicator />}
      {phase === "think" && (
        <div className="self-end max-w-[85%] border-2 border-dashed border-ink rounded-card px-3 py-2 italic text-sm ml-auto"
          style={{ background: "rgba(255,208,41,0.18)" }}>
          <span className="text-2xs not-italic font-display text-ink-soft block mb-0.5">Bạn đang nghĩ</span>
          {line.think}
        </div>
      )}

      {showComposer && (
        <Composer chunks={line.chunks} filled={filled}
          label={phase === "sending" ? "Đang gửi…" : "Soạn tin"} glowIndex={glowIndex} />
      )}
      {showBoard && (
        <div className="grid grid-cols-2 gap-2.5">
          {remaining.map(t => (
            <button key={t.key} data-key={t.key} onClick={() => onTap(t)} disabled={locked}
              className={`relative ${t.side === "en" ? "bg-en" : "bg-vi"} text-white border-2 border-ink rounded-tile px-2.5 pt-4 pb-2 min-h-[56px] text-sm font-display font-bold leading-tight
                focus-visible:outline focus-visible:outline-2 focus-visible:outline-ink
                ${selected?.key === t.key ? "scale-105 ring-2 ring-ink brightness-110" : ""}`}
              style={shaking.includes(t.key) ? { animation: "shake .42s" } : undefined}>
              <span className="absolute top-1 left-2 text-[9px] uppercase opacity-80">{t.side}</span>
              {t.text}
            </button>
          ))}
        </div>
      )}

      {phase === "done" && (
        <div className="flex flex-col gap-3">
          <div className="border-l-4 border-ink rounded-card px-3 py-2 text-sm" style={{ background: "rgba(0,178,169,0.12)" }}>
            <span className="mr-1">🏷️</span>{line.tone}
          </div>
          {line.variants.length > 0 && (
            <div className="border-2 border-ink rounded-card bg-surface">
              <button onClick={() => setVariantOpen(o => !o)} className="w-full text-left px-3 py-2 font-display font-bold text-sm">
                💬 Hoặc người bản xứ còn nói {variantOpen ? "▾" : "▸"}
              </button>
              {variantOpen && (
                <div className="px-3 pb-3 flex flex-col gap-2">
                  {line.variants.map((v, i) => (
                    <div key={i} className="flex flex-col gap-1">
                      <button onClick={() => speak(v.en, { rate: 1 })}
                        className="text-left border-2 border-ink rounded-tile bg-en text-white px-2.5 py-1.5 text-sm font-display font-bold">
                        🔊 {v.en}
                      </button>
                      <p className="text-xs text-ink-soft">{v.why}</p>
                    </div>
                  ))}
                </div>
              )}
            </div>
          )}
          <div className="flex flex-col gap-2 items-stretch">
            <Button onClick={onAdvance}>{state.lineIndex === scene.lines.length - 1 ? "Thử thách cuối ▸" : "Tin tiếp theo ▸"}</Button>
            <Button variant="secondary" onClick={() => speak(line.en, { rate: 0.9 })}>🔊 Nghe lại</Button>
          </div>
        </div>
      )}

      {state.combo >= 2 && showBoard && <ComboBadge combo={state.combo} />}
      {burst && <Particles x={burst.x} y={burst.y} color={burst.color} />}
    </div>
  );
}
```

```tsx
"use client";
import { useMemo, useState } from "react";
import Button from "../ui/Button";
import { speak } from "@/lib/speech";
import { sfx } from "@/lib/sfx";
import type { Scene } from "@/content/types";
import type { GameState } from "@/lib/useGame";

export default function BossScreen({ scene, state, onAnswer, onNext }:
  { scene: Scene; state: GameState; onAnswer: (ok: boolean) => void; onNext: () => void }) {
  const q = scene.boss[state.bossIndex];
  const opts = useMemo(() => [...q.opts].sort(() => Math.random() - 0.5), [state.bossIndex]); // shuffle once
  const [picked, setPicked] = useState<number | null>(null);

  function pick(i: number) {
    if (picked !== null) return;
    setPicked(i);
    const chosen = opts[i];
    const correct = scene.boss[state.bossIndex].opts.find(o => o.ok)!;
    if (chosen.ok) { sfx.win(); speak(chosen.en, { rate: 1 }); onAnswer(true); }
    else { sfx.matchNo(); speak(correct.en, { rate: 1 }); onAnswer(false); }
  }

  return (
    <div className="mx-auto max-w-[520px] px-4 pt-6 pb-8 flex flex-col gap-4">
      <p className="text-2xs font-display text-ink-soft uppercase">Thử thách · {state.bossIndex + 1}/3</p>
      <h2 className="font-display font-bold text-lg">{q.q}</h2>
      <p className="text-xs text-ink-soft">Hai câu này cùng một nghĩa. Câu nào hợp hơn?</p>
      <div className="flex flex-col gap-3">
        {opts.map((o, i) => {
          const revealed = picked !== null;
          const border = !revealed ? "border-2 border-ink"
            : o.ok ? "border-2 border-ok"
            : i === picked ? "border-2 border-dashed border-ink-soft"
            : "border-2 border-ink opacity-60";
          return (
            <div key={i}>
              <button onClick={() => pick(i)} disabled={revealed}
                className={`w-full text-left rounded-tile bg-surface px-3 py-3 font-display font-bold text-sm ${border}
                  focus-visible:outline focus-visible:outline-2 focus-visible:outline-ink`}>
                {o.en}
              </button>
              {revealed && (i === picked || o.ok) && (
                <p className="text-xs text-ink-soft mt-1 px-1">{o.why}</p>
              )}
            </div>
          );
        })}
      </div>
      {picked !== null && (
        <Button onClick={onNext}>{state.bossIndex === scene.boss.length - 1 ? "Xem kết quả ▸" : "Câu tiếp ▸"}</Button>
      )}
    </div>
  );
}
```

"use client";
import { useCallback, useState } from "react";
import scene from "@/content/scenes/sep-hoi-task.json";
import type { Scene } from "@/content/types";
import { useGame } from "@/lib/useGame";
import NameScreen from "@/components/screens/NameScreen";
import IntroScreen from "@/components/screens/IntroScreen";
import PlayScreen from "@/components/screens/PlayScreen";
import BossScreen from "@/components/screens/BossScreen";
import SummaryScreen from "@/components/screens/SummaryScreen";
import ChatThread, { type ChatEntry } from "@/components/game/ChatThread";

const S = scene as Scene;

export default function Page() {
  const g = useGame(S);
  const [chat, setChat] = useState<ChatEntry[]>([]);
  const pushChat = useCallback((e: ChatEntry) => setChat(prev => prev.some(x => x.id === e.id) ? prev : [...prev, e]), []);
  const resetChat = useCallback(() => setChat([]), []);

  switch (g.state.screen) {
    case "name":
      return <NameScreen onSubmit={g.setName} />;
    case "intro":
      return <IntroScreen scene={S} onOpen={() => { resetChat(); g.startScene(); }} />;
    case "play":
      return (
        <PlayScreen scene={S} state={g.state}
          onMatch={g.match} onMiss={g.miss} onAdvance={g.advanceLine}
          onExit={() => { resetChat(); /* back to intro */ location.reload(); }}
          chat={chat} pushChat={pushChat}
          ChatSlot={<ChatThread entries={chat} />} />
      );
    case "boss":
      return <BossScreen scene={S} state={g.state} onAnswer={g.answerBoss} onNext={g.advanceBoss} />;
    case "summary":
      return <SummaryScreen scene={S} state={g.state} chat={chat} onReplay={() => { resetChat(); g.replay(); }} />;
  }
}
